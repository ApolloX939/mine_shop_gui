[General]

# Which currency the shop uses.
# 'auto' picks the first one detected, in this order:
# currency, unified_money, credits, items.
# 'items' uses the item currency configured below.
mine_shop_gui_currency (Currency) enum auto auto,currency,unified_money,credits,items

# Fraction of the buy price that a player receives when selling (0.0 - 1.0).
mine_shop_gui_sell_ratio (Sell price ratio) float 0.5 0.0 1.0

# How much of each currency one price unit is worth. One price unit is the
# value of a cobblestone. Amounts are in the currency's smallest unit
# (cents for currency, whole credits for credits, one item for the item currency).
mine_shop_gui_scale_currency (Scale: currency, cents per unit) float 5 0.0001
mine_shop_gui_scale_unified_money (Scale: unified_money) float 1 0.0001
mine_shop_gui_scale_credits (Scale: credits) float 1 0.0001
mine_shop_gui_scale_items (Scale: item currency) float 0.05 0.0001

[Item currency]

# Use ordinary items as money, for example emeralds under Mineclonia or VoxeLibre.
# Comma-separated list of itemname=value pairs. Values are whole numbers of the
# smallest unit. List larger denominations too if you want change to be made in them.
# Example: mcl_core:emerald_block=9, mcl_core:emerald=1
mine_shop_gui_item_currency (Item currency) string

# Label shown after amounts when the item currency is in use, for example "Emeralds".
