[General]

# Which currency the shop uses.
# 'auto' picks the first one detected, in this order:
# currency, cookies, unified_money, credits, items.
# 'items' uses the item currency configured below.
mine_shop_gui_currency (Currency) enum auto auto,currency,cookies,unified_money,credits,items

# Fraction of the buy price that a player receives when selling (0.0 - 1.0).
mine_shop_gui_sell_ratio (Sell price ratio) float 0.5 0.0 1.0

# How much of each currency one price unit is worth. One price unit is the
# value of a cobblestone. Amounts are in the currency's smallest unit
# (cents for currency, whole credits for credits, one item for the item currency).
mine_shop_gui_scale_currency (Scale: currency, cents per unit) float 5 0.0001
mine_shop_gui_scale_unified_money (Scale: unified_money) float 1 0.0001
mine_shop_gui_scale_credits (Scale: credits) float 1 0.0001

[Shop]

# The most items a player can buy in one go.
mine_shop_gui_max_amount (Largest purchase) int 1000 1 65535

[Auction house]

# The most offers one player can have running at the same time.
mine_shop_gui_ah_max_offers (Offers per player) int 10 1 1000

# How long a new offer runs by default, in hours.
mine_shop_gui_ah_duration (Default offer length, hours) int 24 1 168

# The longest an offer may run, in hours. Players choose from 1, 6, 12, 24, 48, 72 and 168 up to this.
mine_shop_gui_ah_max_hours (Longest offer, hours) int 168 1 168

# How much more than the last bid a new bid must be, in percent (never less than the smallest coin).
mine_shop_gui_ah_bid_step (Bid step, %) float 5 0 100

[Auction house: Discord]

# Post auction offers to Discord. Paste the channel's webhook URL (Channel settings,
# Integrations, Webhooks). Leave empty to turn this off. Treat the URL as a secret.
# The mod also has to be allowed to use the internet: add mine_shop_gui to
# secure.http_mods in minetest.conf.
mine_shop_gui_discord_webhook (Discord webhook URL) string

# One thread per offer, with bids and the result posted into it. This needs the
# webhook's channel to be a Forum channel. Turn it off for an ordinary text
# channel: every message is then posted to the channel itself.
mine_shop_gui_discord_threads (Post each offer as a thread) bool true

# Decorate the posts with emoji (a price tag for sales, a hammer for bids, a check
# mark for SOLD and so on).
mine_shop_gui_discord_emojis (Use emoji) bool true

# Use your own emoji for some of them: comma-separated key=emoji pairs. Keys: sell,
# bid, goods, price, start, current_bid, buyout, ends, ended, sold, won, expired,
# taken_down. A custom server emoji is written <:name:id> (type \:name: in Discord
# with a backslash in front to see it). Example: sold=<:sold:123456789012345678>
mine_shop_gui_discord_emoji_overrides (Emoji overrides) string

# The name the posts appear under.
mine_shop_gui_discord_name (Discord sender name) string Auction House

[Item currency]

# Use ordinary items as money, for example emeralds under Mineclonia or VoxeLibre.
# To use it, also set the currency above to 'items' (or leave it on 'auto').
# Comma-separated list of itemname=value pairs. Values are whole numbers of the
# smallest unit; an item with no value counts as 1. List larger denominations too
# if you want change to be made in them.
# Examples: default:steel_ingot
#           mcl_core:emerald_block=9, mcl_core:emerald=1
mine_shop_gui_item_currency (Item currency) string

# How many price units one unit of the item currency is worth (1 unit = one
# cobblestone). For example 8 means one steel ingot buys what 8 cobblestones cost.
# Leave empty to use the shop's own price for the smallest currency item.
# The items of the currency themselves are never sold in the shop.
mine_shop_gui_item_currency_worth (Item currency worth, price units) float

# Label shown after amounts when the item currency is in use, for example "Emeralds".
mine_shop_gui_item_currency_label (Item currency label) string
